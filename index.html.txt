        async function fetchData() {
            // 주소와 키를 하나로 합친 완성본입니다. (오타 방지용)
            const url = "https://apis.data.go.kr/1471000/MdcinPrdctnImportAcmsltService02/getMdcinPrdctnImportAcmsltList02?serviceKey=3358d0a8ad766199bceba04104a7d6ff5c9851fd5a9de8a01f81ffa06a76edd8&type=json&numOfRows=10&pageNo=1";

            const loading = document.getElementById('loading');
            const table = document.getElementById('resultTable');
            const tbody = document.getElementById('resultBody');

            loading.style.display = 'block';
            table.style.display = 'none';
            tbody.innerHTML = '';

            try {
                // 브라우저 보안(CORS) 문제를 피하기 위해 깃허브 페이지 주소에서 실행해야 합니다.
                const response = await fetch(url);
                const data = await response.json();
                
                if (data.response && data.response.body && data.response.body.items) {
                    const items = data.response.body.items;
                    const itemList = Array.isArray(items) ? items : [items];

                    itemList.forEach(item => {
                        const row = `<tr>
                            <td>${item.ENTP_NAME || '-'}</td>
                            <td>${item.PRDUCT_NAME || '-'}</td>
                            <td>${item.PRDCTN_YEAR || '-'}</td>
                            <td>${item.PRDCTN_AMOUNT ? Number(item.PRDCTN_AMOUNT).toLocaleString() : '0'}</td>
                        </tr>`;
                        tbody.innerHTML += row;
                    });

                    loading.style.display = 'none';
                    table.style.display = 'table';
                } else {
                    // 서버 응답 메시지 확인
                    const msg = data.response ? data.response.header.resultMsg : "데이터 없음";
                    throw new Error(msg);
                }
            } catch (error) {
                loading.style.display = 'none';
                console.error("에러 내용:", error);
                alert("실패 사유: " + error.message + "\n\n※ 만약 'Failed to fetch'라면 브라우저 보안 문제입니다. 깃허브 링크로 접속해 보세요.");
            }
        }
    </script>
